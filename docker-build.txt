LẦN ĐẦU
# 1. Tạo và khởi chạy container lần đầu
docker run -d --name Fermata andreypavlenko/fermata tail -f /dev/null
# 2. Chạy lệnh build bên trong container đó
docker exec -it Fermata ./gradlew bundleAutoRelease -PCUSTOM_APP_ID=com.your.package.name
# 3. Copy file ra ngoài máy thật
docker cp Fermata:/home/circleci/Fermata/fermata/build/outputs/bundle/autoRelease/app-auto-release.aab .


LẦN SAU
# 1. Bật lại container cũ (nếu nó đang tắt), Docker sẽ giữ nguyên mọi cache cũ
docker start Fermata

# 2. Đồng bộ code mới sửa vào container (nếu bạn sửa code ở máy thật)
docker exec -it Fermata /bin/bash
- nếu chưa có git thì
rm -rf Fermata
git clone -b dev https://github.com/hyquoccuong/Fermata.git
- có rồi thì pull
cd Fermata
git pull origin dev

# 3. Ra lệnh build lại. Gradle và CMake thấy cache cũ sẽ nhảy cóc cực nhanh!
docker exec -it Fermata ./gradlew bundleAutoRelease -PCUSTOM_APP_ID=com.your.package.name

# 4. Lấy file .aab mới ra
docker cp Fermata:/home/circleci/Fermata/fermata/build/outputs/bundle/autoRelease/app-auto-release.aab .

Đổi signed key
Mục số 1: Bấm Download encryption public key (Bạn sẽ nhận được file encryption_public_key.pem).
Mục số 2: Bấm Download PEPK tool (Bạn sẽ nhận được file pepk.jar).
Sau khi tải xong, hãy copy cả 2 file encryption_public_key.pem và pepk.jar thảy chung vào thư mục D:\android_keystore\key cho dễ làm việc.

D:\Fermata>docker cp D:\android_keystore\key\. Fermata:/home/circleci/project/
Successfully copied 9.14MB (transferred 9.14MB) to Fermata:/home/circleci/project/

D:\>docker exec -it Fermata /bin/bash
java -jar pepk.jar --keystore=key0 --alias=key0 --output=output.zip --include-cert --rsa-aes-encryption --encryption-key-path=encryption_public_key.pem

D:\Fermata>docker cp Fermata:/home/circleci/project/output.zip D:\android_keystore\key\
Successfully copied 2.62kB (transferred 4.61kB) to D:\android_keystore\key\
