LẦN ĐẦU
# 1. Tạo và khởi chạy container lần đầu
docker run -d --name Fermata andreypavlenko/fermata tail -f /dev/null
sudo apt-get update && sudo apt-get install -y ninja-build
# 2. Chạy lệnh build bên trong container đó
docker exec -it Fermata ./gradlew bundleAutoRelease -PCUSTOM_APP_ID=com.your.package.name
# 3. Copy file ra ngoài máy thật
docker cp Fermata:/home/circleci/Fermata/fermata/build/outputs/bundle/autoRelease/app-auto-release.aab .


LẦN SAU
# 1. Bật lại container cũ (nếu nó đang tắt), Docker sẽ giữ nguyên mọi cache cũ
docker start Fermata

# 2. Đồng bộ code mới sửa vào container (nếu bạn sửa code ở máy thật)
docker exec -it Fermata /bin/bash
- nếu chưa có git thì
rm -rf Fermata
git clone -b dev https://github.com/hyquoccuong/Fermata.git
- có rồi thì pull
cd Fermata
git pull origin dev

# 3. Ra lệnh build lại. Gradle và CMake thấy cache cũ sẽ nhảy cóc cực nhanh!
docker exec -it Fermata ./gradlew bundleAutoRelease -PCUSTOM_APP_ID=com.your.package.name
thêm --configuration-cache để tăng tốc https://docs.gradle.org/9.5.1/userguide/configuration_cache_enabling.html
docker exec -it Fermata ./gradlew bundleAutoRelease -PCUSTOM_APP_ID=com.hytruclinh.apptool.beta

# 4. Lấy file .aab mới ra
docker cp Fermata:/home/circleci/Fermata/fermata/build/outputs/bundle/autoRelease/app-auto-release.aab .
